⚡ JavaScript

JavaScript es el lenguaje de programación del navegador. Hace las páginas web dinámicas e interactivas: responde a clics, valida formularios, carga datos sin recargar la página y mucho más.

1. ¿Qué es JavaScript?

JavaScript (JS) es un lenguaje de programación interpretado, orientado a objetos y dinámico. A diferencia de HTML y CSS, JS sí tiene lógica: puede tomar decisiones, repetir acciones, calcular valores y modificar la página en tiempo real.

Se ejecuta directamente en el navegador del usuario (lado cliente), aunque también puede ejecutarse en el servidor con Node.js. En esta guía nos centraremos en el uso cliente.

📌 Cómo incluir JS en HTML 
<!-- Opción 1: JS externo (recomendado) — colocar antes de </body> -->
<script src="js/mi-script.js"></script>

<!-- Opción 2: JS inline (solo para pruebas pequeñas) -->
<script>
  console.log("Hola desde JavaScript");
</script>

<!--
  Nota: colocar el <script> antes de </body> asegura que el HTML ya está
  cargado cuando el JS se ejecuta. También puedes usar defer:
-->
<script src="js/mi-script.js" defer></script>

2. Variables: var, let y const

Una variable es un contenedor con nombre para almacenar un valor. En JavaScript moderno tenemos tres formas de declarar variables: 

// ── const: valor constante (NO puede reasignarse) ── ✅ usar por defecto
const PI = 3.14159;
const nombre = "Ana García";
const url = "https://api.ejemplo.com/datos";

// const NO significa que el objeto sea inmutable:
const persona = { nombre: "Ana" };
persona.nombre = "Carlos";  // Esto SÍ funciona (modificar propiedad)
// persona = {};              // Esto NO funciona (reasignar variable)

// ── let: valor que puede cambiar ── ✅ usar cuando necesitas cambiar el valor
let contador = 0;
let estaConectado = true;

contador = contador + 1;   // Permitido: reasignación
contador++;                // Atajo: incrementar en 1
estaConectado = false;     // Permitido: cambio de valor

// ── var: forma antigua (evitar en código moderno) ── ❌ no usar
var antiguo = "evitar";
// Problemas de var:
// - Tiene scope de función, no de bloque (comportamiento confuso)
// - Se "iza" (hoisting) al inicio de la función
// - Permite redeclaración accidental
Palabra claveReasignableScopeRecomendación
constNoBloque✅ Usar siempre por defecto
letSíBloque✅ Usar si necesitas reasignar
varSíFunción❌ Evitar

4. Funciones

// ── Declaración de función (function declaration) ──
// Se puede llamar antes de ser definida (hoisting)
function sumar(a, b) {
  return a + b;
}
console.log(sumar(3, 5)); // → 8

// ── Expresión de función (function expression) ──
const multiplicar = function(a, b) {
  return a * b;
};
console.log(multiplicar(3, 5)); // → 15

// ── Arrow function (función flecha) ── ES6+ ✅ recomendada
const dividir = (a, b) => {
  if (b === 0) return "Error: división por cero";
  return a / b;
};

// Si solo hay una expresión, puedes omitir {} y return:
const cuadrado   = x => x * x;
const sumarFlecha = (a, b) => a + b;

console.log(cuadrado(4));      // → 16
console.log(sumarFlecha(3, 5)); // → 8

// ── Parámetros con valor por defecto ──
function saludar(nombre = "estudiante", saludo = "Hola") {
  return `${saludo}, ${nombre}!`;
}
console.log(saludar());              // → "Hola, estudiante!"
console.log(saludar("Ana"));         // → "Hola, Ana!"
console.log(saludar("Ana", "Buenos días")); // → "Buenos días, Ana!"

// ── Rest parameters (número variable de argumentos) ──
function sumarTodos(...numeros) {
  return numeros.reduce((total, n) => total + n, 0);
}
console.log(sumarTodos(1, 2, 3, 4, 5)); // → 15

// ── Callbacks: funciones pasadas como argumento ──
function procesarNumeros(array, callback) {
  return array.map(callback);
}
const dobles = procesarNumeros([1, 2, 3], n => n * 2);
console.log(dobles); // → [2, 4, 6]

// setTimeout usa un callback (ejecuta la función después de N ms)
setTimeout(function() {
  console.log("Han pasado 2 segundos");
}, 2000);

7. Fetch API — Peticiones al servidor

La API Fetch permite hacer peticiones HTTP desde JavaScript sin recargar la página (AJAX moderno). Es la forma estándar de comunicarse con el servidor o con APIs externas. 

// ── Fetch básico con .then() ──
fetch('https://api.ejemplo.com/usuarios')
  .then(function(response) {
    // Comprobar si la respuesta fue exitosa
    if (!response.ok) {
      throw new Error('Error HTTP: ' + response.status);
    }
    // Convertir la respuesta a JSON
    return response.json();
  })
  .then(function(datos) {
    // Aquí tenemos los datos parseados
    console.log('Usuarios recibidos:', datos);
    mostrarUsuarios(datos);
  })
  .catch(function(error) {
    // Capturar cualquier error (red, HTTP, JSON inválido...)
    console.error('Error al cargar usuarios:', error);
    mostrarError('No se pudieron cargar los usuarios.');
  });

// ── Fetch moderno con async/await ── ✅ más legible
async function cargarUsuarios() {
  try {
    const response = await fetch('https://api.ejemplo.com/usuarios');

    if (!response.ok) {
      throw new Error(`Error HTTP: ${response.status}`);
    }

    const usuarios = await response.json();
    return usuarios;

  } catch (error) {
    console.error('Error:', error);
    throw error; // Re-lanzar para que lo maneje quien llamó a la función
  }
}

// Llamar a la función async
cargarUsuarios()
  .then(usuarios => {
    console.log(usuarios);
    // Mostrar usuarios en el DOM
    const lista = document.querySelector('#lista-usuarios');
    lista.innerHTML = usuarios.map(u => `
      <li class="usuario">
        <strong>${u.nombre}</strong> — ${u.email}
      </li>
    `).join('');
  })
  .catch(err => console.error(err));

// ── POST: enviar datos al servidor ──
async function crearUsuario(nombre, email) {
  try {
    const response = await fetch('/api/usuarios', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json', // Indicamos que enviamos JSON
      },
      body: JSON.stringify({ nombre, email }) // Convertir objeto a string JSON
    });

    if (!response.ok) {
      throw new Error(`Error: ${response.status}`);
    }

    const nuevoUsuario = await response.json();
    console.log('Usuario creado:', nuevoUsuario);
    return nuevoUsuario;

  } catch (error) {
    console.error('Error al crear usuario:', error);
  }
}

// Ejemplo: enviar formulario con Fetch
document.querySelector('#form-usuario').addEventListener('submit', async (e) => {
  e.preventDefault();

  const nombre = document.querySelector('#nombre').value;
  const email  = document.querySelector('#email').value;

  const usuario = await crearUsuario(nombre, email);

  if (usuario) {
    // Mostrar mensaje de éxito
    document.querySelector('#mensaje').textContent = '¡Usuario creado con éxito!';
    e.target.reset(); // Limpiar formulario
  }
});

// ✅ Separar responsabilidades
function validarNombre(nombre) {
  return nombre.trim().length >= 2;
}

async function guardarEnServidor(datos) {
  const res = await fetch('/api/guardar', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(datos)
  });
  return res.json();
}

function mostrarConfirmacion() {
  document.querySelector('.confirmacion').style.display = 'block';
}

async function manejarEnvioFormulario(e) {
  e.preventDefault();
  const nombre = e.target.nombre.value;

  if (!validarNombre(nombre)) {
    mostrarError('Nombre demasiado corto');
    return;
  }

  await guardarEnServidor({ nombre });
  mostrarConfirmacion();
}
